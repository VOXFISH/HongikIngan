# Girl%d — 이름

> 이 문서를 `Girl1.md`, `Girl2.md`처럼 복사하고 제목·파일명·식별자의 `%d`를 같은 정수로 바꾼다.
> 위쪽은 JSON 모양으로 정보를 정리하고, 아래쪽은 설정·스킬 동작·기타 내용을 자유롭게 적는다. 엄격한 JSON 문법이나 고정된 형식에 맞출 필요는 없다.
> 빈칸과 `미정`은 아직 정하지 않은 내용이다. AI는 코드 블록과 아래 설명을 모두 읽고, 모순되거나 미정인 내용은 임의로 확정하지 않는다.

시너지의 공통 효과는 [시너지 목록](synergy.md), 키워드의 공통 효과는 [키워드 목록](keywords.md)에 적고 이 문서에서는 같은 식별자나 이름으로 참조한다.

## 기본 정보

```json
{
  "서식 버전": 1,
  "마법소녀 식별자": "Girl%d",
  "작성 상태": "작성 중", // 작성 중 / 구현 준비 완료
  "이름": "",
  "소개": "", // 성격, 배경, 전투 컨셉을 자유롭게 설명
  "학년": "", // 1, 2, 3 중 하나. 가격은 학년별 공통 경제 설정을 따름
  "표시 역할군": "", // 지원 / 약화 / 공격. 화면 표시용이며 효과 조건으로 사용하지 않음
  "시너지": {
    "부활동": "", // 시너지 목록의 식별자나 이름
    "성격": "",
    "특수 시너지": [], // 없으면 빈 목록으로 둠
    "추가 설명": "" // 공통 효과와 다른 개별 동작이나 연결 의도
  },
  "스킬": {
    "스킬 식별자": "Girl%d_스킬",
    "이름": "",
    "표시 설명": "", // 현재 수치를 넣을 자리는 {효과값}처럼 표기
    "기획 의도": "", // 어떤 조합과 배치에서 어떤 플레이를 유도하는지
    "수치 목록": [
      { "수치 이름": "효과값", "초기값": "", "단위": "", "설명": "" }
    ],
    "강화": {
      "방식": "", // 1회만 가능 / 무제한 가능
      "강화할 수치": "효과값", // 수치 목록에서 하나만 지정
      "새 값 계산 방법": "", // 강화 후 횟수와 현재 값을 어떻게 사용하는지 설명. 결과로 교체하며 결과를 다시 더하지 않음
      "강화 예시": [
        { "강화 횟수": 0, "예상 값": "" },
        { "강화 횟수": 1, "예상 값": "" },
        { "강화 횟수": 2, "예상 값": "" }
      ] // 1회형은 2회 항목 삭제. 무제한형은 이후 횟수에도 적용할 계산 방법을 설명
    },
    "동작 규칙": [
      {
        "규칙 식별자": "Girl%d_규칙1",
        "발동 계기": "", // 예: 배치 시 / 자기 슬롯 행동 / 다른 아군 행동 / 이동 시
        "발동 조건": "", // 무조건이면 '없음'. 조건을 확인하는 시점도 설명
        "실행 순서대로 적은 효과": [
          {
            "설명": "", // 효과가 여러 개면 이 항목을 복사해 실행 순서대로 작성
            "효과 종류": "", // 예: 합동 데미지 더하기 / 비율 더하기 / 곱하기 / 키워드 부여 / 위치 교환 / 재화 획득
            "적용할 수치 또는 계산식": "효과값", // 현재 스킬 수치를 사용. 비율이면 기준값과 단위도 설명
            "대상": {
              "선택 방법": "", // 예: 자신 / 다음 아군 / 다른 아군 전체 / 보스 / 합동 데미지
              "선택 시점": "", // 예: 효과 실행 직전
              "후보가 여럿일 때": "", // 선택 우선순위. 무관하면 '해당 없음'
              "대상이 없을 때": "" // 효과 생략 등 원하는 동작을 명시
            },
            "부여할 키워드": "", // 키워드 부여일 때만 목록의 식별자나 이름 입력
            "부여할 중첩 수": "" // 키워드 부여일 때만 입력
          }
        ],
        "연쇄 반응 설명": "" // 자신에 대한 반응 제외, 다른 반응의 추가 발생, 위치 교환 후 재행동, 무한 반복 가능성 등
      }
    ]
  },
  "사용 키워드": [], // 키워드 목록에 정의한 식별자나 이름. 공통 효과는 그 문서에 작성
  "외형과 연출": {
    "초상화 자료": "", // 기존 이미지 경로 또는 제작 계획. 없으면 비워 둠
    "전투 모습 자료": "",
    "대표 색상": "", // 색 이름이나 색상 코드로 작성
    "연출 설명": "" // 실제 게임 효과와 순수 연출을 구별
  },
  "확인할 동작 예시": [
    {
      "상황 이름": "기본 행동",
      "시작 상황": "", // 배치, 강화 횟수, 현재 데미지, 키워드 등
      "발생한 행동": "",
      "기대 결과": "" // 수치·대상·키워드·슬롯이 어떻게 바뀌는지
    },
    { "상황 이름": "강화 후 행동", "시작 상황": "", "발생한 행동": "", "기대 결과": "" },
    { "상황 이름": "대상 없음 또는 조건 불충족", "시작 상황": "", "발생한 행동": "", "기대 결과": "" }
  ],
  "미정 사항": [], // 결정하지 못한 내용을 적고, 남아 있다면 '구현 준비 완료'로 바꾸지 않음
  "구현자에게 전달할 내용": ""
}
```
